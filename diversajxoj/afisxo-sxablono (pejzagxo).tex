\begin{align*}
    \det \begin{bmatrix}
             a_{1,1} & a_{1,2} & a_{1,3} \\
             a_{2,1} & a_{2,2} & a_{2,3} \\
             a_{3,1} & a_{3,2} & a_{3,3} \\
         \end{bmatrix}
    =
    a_{1,1} a_{2,2} a_{3,3}
    + a_{1,2} a_{2,3} a_{3,1}
    + a_{1,3} a_{2,1} a_{3,2}
    - a_{1,1} a_{2,3} a_{3,2}
    - a_{1,2} a_{2,1} a_{3,3}
    - a_{1,3} a_{2,2} a_{3,1}
\end{align*}
